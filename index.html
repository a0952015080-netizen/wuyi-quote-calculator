    const state = {

    function setFlowerMode(mode) {

      state.flowerMode = mode;

      document.getElementById('btnFlowerNone').classList.toggle('active', mode === 'none');

      document.getElementById('btnFlowerActive').classList.toggle('active', mode === 'flower');

      document.getElementById('flowerDetail').style.display = (mode === 'flower') ? 'block' : 'none';

      calc();

    }


    function setFlowerPriceType(type) {

      state.flowerPriceType = type;

      document.getElementById('btnFlower390').classList.toggle('active', type === '390');

      document.getElementById('btnFlowerCustom').classList.toggle('active', type === 'custom');

      document.getElementById('flowerCustomInputWrap').style.display = (type === 'custom') ? 'block' : 'none';

      if (type === 'custom') {

        document.getElementById('flowerCustomPrice').focus();

      }

      calc();

    }


    function toggleBag() {

      state.hasBag = !state.hasBag;

      document.getElementById('bagToggle').classList.toggle('active', state.hasBag);

      calc();

    }


    function setCakeMode(mode) {

      state.cakeMode = mode;

      document.getElementById('btnCakeNone').classList.toggle('active', mode === 'none');

      document.getElementById('btnCakeActive').classList.toggle('active', mode === 'cake');

      document.getElementById('cakeDetail').style.display = (mode === 'cake') ? 'block' : 'none';

      if (mode === 'cake') {

        document.getElementById('cakePrice').focus();

      }

      calc();

    }


    function toggleAddon(key, name, price) {

      const btn = document.getElementById(`addon-${key}`);

      if (state.addons.has(key)) {

        state.addons.delete(key);

        btn.classList.remove('active');

      } else {

        state.addons.set(key, { name, price });

        btn.classList.add('active');

      }

      calc();

    }


    function selectCandle(price) {

      const btn45 = document.getElementById('addon-candle45');

      const btn55 = document.getElementById('addon-candle55');

      if (state.candlePrice === price) {

        state.candlePrice = 0;

        btn45.classList.remove('active');

        btn55.classList.remove('active');

      } else {

        state.candlePrice = price;

        btn45.classList.toggle('active', price === 45);

        btn55.classList.toggle('active', price === 55);

      }

      calc();

    }


    function changeNumCandle(delta) {

      state.numCandleQty = Math.max(0, state.numCandleQty + delta);

      document.getElementById('numCandleQty').textContent = state.numCandleQty;

      document.getElementById('numCandleCard').classList.toggle('active', state.numCandleQty > 0);

      calc();

    }


    function toggleIceBag() {

      state.hasIceBag = !state.hasIceBag;

    }


    function getSummaryData() {

      const items = [];

      let total = 0;


      // 花束

      if (state.flowerMode === 'flower') {

        let fPrice = 0;

        if (state.flowerPriceType === '390') {

          fPrice = 390;

        } else {

          const val = parseInt(document.getElementById('flowerCustomPrice').value, 10);

          if (!isNaN(val) && val > 0) fPrice = val;

        }


        if (fPrice > 0) {

          items.push(`花束$${fPrice}`);

          total += fPrice;

        }


        if (state.hasBag) {

          items.push('提袋$40');

          total += 40;

        }

      }


      // 蛋糕

      if (state.cakeMode === 'cake') {

        const cPrice = parseInt(document.getElementById('cakePrice').value, 10);

        if (!isNaN(cPrice) && cPrice > 0) {

          items.push(`蛋糕$${cPrice}`);

          total += cPrice;

        }

      }


      // 餐具

      state.addons.forEach(item => {

        items.push(`${item.name}$${item.price}`);

        total += item.price;

      });


      // 蠟燭

      if (state.candlePrice > 0) {

        items.push(`蠟燭$${state.candlePrice}`);

        total += state.candlePrice;

      }


      // 數字蠟燭

      if (state.numCandleQty > 0) {

        const ncPrice = state.numCandleQty * 30;

        items.push(`數字蠟燭(${state.numCandleQty}支)$${ncPrice}`);

        total += ncPrice;

      }


      // 保冰袋

      if (state.hasIceBag) {

        items.push('保冰袋$100');

        total += 100;

      }


      // 其他

      const otherPrice = parseInt(document.getElementById('otherPrice').value, 10);

      if (!isNaN(otherPrice) && otherPrice > 0) {

        const otherName = document.getElementById('otherName').value.trim() || '其他';

        items.push(`${otherName}$${otherPrice}`);

        total += otherPrice;

      }


      // 備註

      const notePrice = parseInt(document.getElementById('notePrice').value, 10);

      if (!isNaN(notePrice) && notePrice > 0) {

        const noteName = document.getElementById('noteName').value.trim() || '備註';

        items.push(`${noteName}$${notePrice}`);

        total += notePrice;

      }


      return { items, total };

    }


    function buildOutputText(items, total) {

      const header = '您好呀～\n幫您加總一下';

      const footer = `總共$${total}\n您看對不對喔～`;

      if (items.length === 0) {

        return `${header}\n${footer}`;

      }

      return `${header}\n${items.join('\n')}\n${footer}`;

    }

      document.getElementById('previewText').textContent = buildOutputText(items, total);

    function copyAndReturn() {

      const { items, total } = getSummaryData();

      if (items.length === 0) {

        alert('請先選擇或輸入至少一項品項金額！');

        return;

      }


      const text = buildOutputText(items, total);


      const handleSuccess = () => {

        showToast('已複製！即將跳回 LINE');

        setTimeout(() => {

          window.location.href = "line://";

        }, 300);

      };


      if (navigator.clipboard && navigator.clipboard.writeText) {

        navigator.clipboard.writeText(text).then(handleSuccess).catch(() => {

          fallbackCopy(text);

          handleSuccess();

        });

      } else {

        fallbackCopy(text);

        handleSuccess();

      }

    }


    function fallbackCopy(text) {

      const ta = document.createElement("textarea");

      ta.value = text;

      ta.style.position = 'fixed';

      ta.style.opacity = '0';

      document.body.appendChild(ta);

      ta.focus();

      ta.select();

      try {

        document.execCommand('copy');

      } catch (e) {

        console.error('複製失敗', e);

      }

      document.body.removeChild(ta);

    }
